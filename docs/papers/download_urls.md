# 論文PDF取得用URL一覧

以下のコマンドをローカル環境で実行してください。

```bash
# 最優先4本（動機×誘因の交差点）
curl -L -o deci_koestner_ryan_1999.pdf "https://selfdeterminationtheory.org/wp-content/uploads/2014/04/1999_DeciKoestnerRyan_Meta.pdf"
curl -L -o cerasoli_nicklin_ford_2014.pdf "https://selfdeterminationtheory.org/wp-content/uploads/2017/06/2014_Cerasoli_Intrinsic.pdf"
curl -L -o benabou_tirole_2003.pdf "https://www.princeton.edu/~rbenabou/papers/RES2003.pdf"
curl -L -o frey_jegen_2001.pdf "https://clivespash.org/speer/iewwp026.pdf"

# 高優先（理論的基盤）
curl -L -o gagne_deci_2005.pdf "https://selfdeterminationtheory.org/SDT/documents/2005_GagneDeci_JOB_SDTtheory.pdf"
curl -L -o gneezy_rustichini_2000.pdf "https://rady.ucsd.edu/faculty/directory/gneezy/pub/docs/fine.pdf"
curl -L -o ryan_deci_2000.pdf "https://selfdeterminationtheory.org/SDT/documents/2000_RyanDeci_SDT.pdf"

# ユーザ提供URL
curl -L -o deci_koestner_ryan_1999_alt.pdf "https://home.ubalt.edu/tmitch/642/articles%20syllabus/Deci%20Koestner%20Ryan%20meta%20IM%20psy%20bull%2099.pdf"
```
